cmake_minimum_required(VERSION 3.21)
project(OrbisHost VERSION 0.1.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Network)
qt_standard_project_setup(REQUIRES 6.8)
set(ORBIS_DEFAULT_RELAY "wss://orbising.com/relay" CACHE STRING "Default registration and Relay server")
qt_add_executable(OrbisHost WIN32 src/main.cpp src/HostController.cpp src/HostController.h src/CredentialStore.cpp src/CredentialStore.h)
if(WIN32)
  target_sources(OrbisHost PRIVATE src/app.rc)
  target_link_libraries(OrbisHost PRIVATE crypt32)
endif()
set_source_files_properties(src/qml/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
set_source_files_properties(src/qml/Card.qml PROPERTIES QT_RESOURCE_ALIAS Card.qml)
set_source_files_properties(src/qml/ActionButton.qml PROPERTIES QT_RESOURCE_ALIAS ActionButton.qml)
qt_add_qml_module(OrbisHost URI Orbis.Host VERSION 1.0 QML_FILES src/qml/Main.qml src/qml/Card.qml src/qml/ActionButton.qml SOURCES src/NeuSurface.cpp src/NeuSurface.h)
qt_add_resources(OrbisHost branding PREFIX "/" FILES src/assets/orbis.png)
target_compile_definitions(OrbisHost PRIVATE ORBIS_VERSION="${PROJECT_VERSION}" ORBIS_DEFAULT_RELAY="${ORBIS_DEFAULT_RELAY}")
target_link_libraries(OrbisHost PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Network)
target_include_directories(OrbisHost PRIVATE src)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(credential-store-test src/tests/CredentialStoreTest.cpp src/CredentialStore.cpp)
  target_link_libraries(credential-store-test PRIVATE Qt6::Core Qt6::Test crypt32)
  add_test(NAME credential-store COMMAND credential-store-test)
endif()
